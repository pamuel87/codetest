📘 Firebase / GitHub Pages 세팅 가이드 & Security Rules

1. 교사 계정 사전 등록 방법 (Firebase Console)
  1. Firebase Authentication 사용자 목록에서 교사의 Google 계정 UID를 복사합니다.
  2. Firestore Database에서 /roles/{UID} 문서를 직접 생성합니다.
  3. 문서 필드 추가: role: "teacher" (string), status: "approved" (string), classId: "all" (string).
2. Cloud Firestore Security Rules (보안 규칙)
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper function for Auth check
    function isAuthenticated() {
      return request.auth != null;
    }
    function isApprovedTeacher() {
      return isAuthenticated() && 
        get(/databases/$(database)/documents/roles/$(request.auth.uid)).data.role == 'teacher' &&
        get(/databases/$(database)/documents/roles/$(request.auth.uid)).data.status == 'approved';
    }

    // Role management
    match /roles/{userId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.auth.uid == userId;
      allow update: if isApprovedTeacher();
    }

    // Student Submissions
    match /submissions/{subId} {
      allow read: if isAuthenticated() && (resource.data.studentUid == request.auth.uid || isApprovedTeacher());
      allow create: if isAuthenticated() && request.resource.data.studentUid == request.auth.uid;
      allow update: if isApprovedTeacher();
    }
  }
}
